    04 · Logika diskon
    
      
    Diskon dihitung dari lama menginap
    
      
    Menginap 3 hari mendapat diskon 10%, 7 hari mendapat diskon 20%. Total akhir dipakai untuk menghitung kembalian.
    
      
    Lihat full code
    
    
    
    
    int diskonPersen = 0;
if (durasi >= 7) {
    diskonPersen = 20;
} else if (durasi >= 3) {
    diskonPersen = 10;
}

long diskonNominal = (totalAwal * diskonPersen) / 100;
long totalAkhir = totalAwal - diskonNominal;
